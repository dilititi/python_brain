      rows = [
          {"name": " Ada ", "score": "98"},
          {"name": "", "score": "75"},
          {"name": "Guido", "score": "91"},
      ]

      cleaned = [
          {"name": row["name"].strip(), "score": int(row["score"])}
          for row in rows
          if row["name"].strip()
      ]

      print(sorted(cleaned, key=lambda row: row["score"], reverse=True))
