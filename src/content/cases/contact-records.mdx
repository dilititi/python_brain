codeVersions:

    code: |
      from dataclasses import dataclass

      @dataclass
      class Record:
          name: str

          def label(self):
              return self.name

      @dataclass
      class Contact(Record):
          email: str
          tags: list[str]

          def __str__(self):
              return f"{self.name} <{self.email}>"

          def matches(self, tag):
              return tag in self.tags

      contacts = [
          Contact("Ada", "ada@example.com", ["vip"]),
          Contact("Guido", "guido@example.com", ["python"]),
      ]

      for contact in contacts:
          if contact.matches("python"):
              print(contact.label(), str(contact))
pitfalls:
  - 联系人字段散落在 dict 中时，拼写错误和缺字段通常到运行时才暴露。
extensions:
  - 用 dataclass 增加字段默认值、格式化输出和 JSON 持久化。
