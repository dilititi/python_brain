codeExamples:
  - title: naive
    description: 遍历列表时原地删除元素，容易因为位置移动而漏掉数据。
    code: |
      scores = [10, -1, -2, 5]

      for score in scores:
          if score < 0:
              scores.remove(score)

      print(scores)
  - title: standard
    description: 需要过滤列表时，构造一个新列表通常更清楚。
    code: |
      scores = [10, -1, -2, 5]
      valid_scores = []

      for score in scores:
          if score >= 0:
              valid_scores.append(score)

      print(valid_scores)
  - title: production
    description: 工程版把列表处理写成函数，并用边界检查拒绝不合理分数。
    code: |
      from typing import Iterable

      def clean_scores(scores: Iterable[int]) -> list[int]:
          valid_scores: list[int] = []

          for score in scores:
              if score > 100:
                  raise ValueError("score cannot exceed 100")
              if score >= 0:
                  valid_scores.append(score)

          return valid_scores

      print(clean_scores([10, -1, -2, 5]))
