  - title: production
    description: 工程版显式解析布尔文本，并对未知值抛出清楚错误。
    code: |
      def parse_bool(raw: str | bool | None) -> bool:
          if raw is None:
              return False
          if isinstance(raw, bool):
              return raw

          normalized = raw.strip().lower()
          if normalized in {"true", "1", "yes"}:
              return True
          if normalized in {"false", "0", "no"}:
              return False
          raise ValueError(f"cannot parse boolean value: {raw}")

      print(parse_bool("False"))
