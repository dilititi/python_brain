codeExamples:
  - title: naive
    description: 初学者常把 or 写成自然语言，导致条件比想象中更宽。
    code: |
      role = "viewer"
      active = True

      if role == "admin" or "editor" and active:
          print("allow")
      else:
          print("deny")
  - title: standard
    description: 用 in 表达成员关系，再用 and 组合必要条件。
    code: |
      role = "viewer"
      active = True

      if role in {"admin", "editor"} and active:
          print("allow")
      else:
          print("deny")
  - title: production
    description: 工程版把规则封装成函数，并把 locked 这类边界条件写进判断。
    code: |
      from typing import Literal

      Role = Literal["admin", "editor", "viewer"]

      def can_edit(role: Role, active: bool, locked: bool) -> bool:
          return role in {"admin", "editor"} and active and not locked

      checks = [
          ("editor", True, False),
          ("viewer", True, False),
          ("admin", True, True),
      ]

      for role, active, locked in checks:
          print(role, can_edit(role, active, locked))
